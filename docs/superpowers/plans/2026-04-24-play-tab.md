# Play Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the existing Match Feed tab with a map-first "Play" tab that surfaces nearby teammate-seeking posts as pins on a map, with a single prominent "I want to play" CTA and a rich create-post flow (Where / When / Format / Spots / Type).

**Architecture:** Reuse the existing backend match_post endpoints + geo filter + FCM push. Add two new columns (`duration_minutes`, `format`) to `match_posts` for richer posts. Replace `feed_screen.dart` with a new `play_screen.dart` that is map-first by default, with a draggable bottom sheet for the nearby list. Router nav label + icon updated. Old feed screen deleted.

**Tech Stack:** Go + GORM + Postgres (backend); Flutter + Riverpod + MapLibre + Freezed (frontend).

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:google_fonts/google_fonts.dart';
import 'package:intl/intl.dart';
import '../../../../core/theme/app_colors.dart';
import '../../../../data/sources/remote/feed_remote_source.dart';
import '../../../providers/feed_provider.dart';

const _kFormats = ['5v5', '6v6', '7v7', '11v11'];

class IWantToPlaySheet extends ConsumerStatefulWidget {
  const IWantToPlaySheet({
    super.key,
    required this.userLat,
    required this.userLng,
  });

  final double? userLat;
  final double? userLng;

  @override
  ConsumerState<IWantToPlaySheet> createState() => _IWantToPlaySheetState();
}

class _IWantToPlaySheetState extends ConsumerState<IWantToPlaySheet> {
  final _titleCtrl = TextEditingController();
  final _noteCtrl = TextEditingController();
  DateTime _matchDate = DateTime.now().add(const Duration(hours: 2));
  int _durationMinutes = 90;
  String _format = '7v7';
  String _postType = 'open_match';
  int _spots = 10;
  bool _submitting = false;

  @override
  void dispose() {
    _titleCtrl.dispose();
    _noteCtrl.dispose();
    super.dispose();
  }

  Future<void> _pickDateTime() async {
    final now = DateTime.now();
    final date = await showDatePicker(
      context: context,
      firstDate: now,
      lastDate: now.add(const Duration(days: 30)),
      initialDate: _matchDate,
      builder: (ctx, child) => Theme(
        data: Theme.of(ctx).copyWith(colorScheme: const ColorScheme.dark(primary: AppColors.primary)),
        child: child!,
      ),
    );
    if (date == null || !mounted) return;
    final time = await showTimePicker(
      context: context,
      initialTime: TimeOfDay.fromDateTime(_matchDate),
      builder: (ctx, child) => Theme(
        data: Theme.of(ctx).copyWith(colorScheme: const ColorScheme.dark(primary: AppColors.primary)),
        child: child!,
      ),
    );
    if (time == null || !mounted) return;
    setState(() => _matchDate = DateTime(date.year, date.month, date.day, time.hour, time.minute));
  }

  Future<void> _submit() async {
    if (_submitting) return;
    setState(() => _submitting = true);
    try {
      final remote = ref.read(feedRemoteSourceProvider);
      await remote.createPost(
        title: _titleCtrl.text.trim().isEmpty ? 'Looking to play' : _titleCtrl.text.trim(),
        description: _noteCtrl.text.trim(),
        matchDate: _matchDate,
        spotsTotal: _spots,
        postType: _postType,
        latitude: widget.userLat,
        longitude: widget.userLng,
        durationMinutes: _durationMinutes,
        format: _format,
      );
      if (!mounted) return;
      ref.invalidate(feedListProvider);
      Navigator.of(context).pop(true);
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:geolocator/geolocator.dart';

import '../../providers/feed_provider.dart';
import 'widgets/i_want_to_play_sheet.dart';
import 'widgets/nearby_posts_sheet.dart';
import 'widgets/play_map_view.dart';
import 'widgets/post_details_card.dart';

class PlayScreen extends ConsumerStatefulWidget {
  const PlayScreen({super.key});

  @override
  ConsumerState<PlayScreen> createState() => _PlayScreenState();
}

class _PlayScreenState extends ConsumerState<PlayScreen> {
  double? _userLat;
  double? _userLng;
  MatchPost? _selected;
  bool _joining = false;

  @override
  void initState() {
    super.initState();
    _getLocation();
  }

  Future<void> _getLocation() async {
    try {
      var perm = await Geolocator.checkPermission();
      if (perm == LocationPermission.denied) {
        perm = await Geolocator.requestPermission();
      }
      if (perm == LocationPermission.denied || perm == LocationPermission.deniedForever) {
        return;
      }
      final pos = await Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(accuracy: LocationAccuracy.low),
      ).timeout(const Duration(seconds: 6));
      if (!mounted) return;
      setState(() { _userLat = pos.latitude; _userLng = pos.longitude; });
    } catch (_) {}
  }

  Future<void> _openCreateSheet() async {
    final created = await showModalBottomSheet<bool>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (_) => IWantToPlaySheet(userLat: _userLat, userLng: _userLng),
    );
    if (created == true && mounted) {
      ref.invalidate(feedListProvider);
    }
  }

  Future<void> _join(MatchPost post) async {
    if (_joining) return;
    setState(() => _joining = true);
    try {
      final remote = ref.read(feedRemoteSourceProvider);
      await remote.joinPost(post.id);
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Joined ${post.title.isEmpty ? "match" : post.title}')),
      );
      ref.invalidate(feedListProvider);
      setState(() { _selected = null; _joining = false; });
    } catch (e) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Could not join: $e')),
      );
      setState(() => _joining = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final feedAsync = ref.watch(feedListProvider(
      lat: _userLat,
      lng: _userLng,
      postType: null,
    ));

    return Scaffold(
      backgroundColor: AppColors.background,
      body: Stack(
        children: [
          // --- Map ---
          Positioned.fill(
            child: feedAsync.when(
              data: (posts) => PlayMapView(
                posts: posts,
                userLat: _userLat,
                userLng: _userLng,
                onPinTap: (p) => setState(() => _selected = p),
              ),
              loading: () => const Center(child: CircularProgressIndicator(color: AppColors.primary)),
              error: (e, _) => Center(
                child: Padding(
                  padding: const EdgeInsets.all(24),
                  child: Text('Could not load posts: $e',
                    style: GoogleFonts.dmSans(color: AppColors.textSecondary),
                    textAlign: TextAlign.center,
                  ),
                ),
              ),
            ),
          ),

          // --- Bottom draggable nearby sheet (peek) ---
          feedAsync.maybeWhen(
            data: (posts) => DraggableScrollableSheet(
              initialChildSize: 0.18,
              minChildSize: 0.12,
              maxChildSize: 0.85,
              builder: (_, ctrl) => NearbyPostsSheet(
                posts: posts,
                scrollController: ctrl,
                onTap: (p) => setState(() => _selected = p),
              ),
            ),
            orElse: () => const SizedBox.shrink(),
          ),

          // --- "I want to play" FAB, anchored above the sheet peek ---
          Positioned(
            left: 16, right: 16, bottom: 150,
            child: SafeArea(
              top: false,
              child: Center(
                child: ElevatedButton.icon(
                  onPressed: _openCreateSheet,
                  icon: const Icon(Icons.sports_soccer_rounded, size: 20),
                  label: Text(
                    'I want to play',
                    style: GoogleFonts.nunito(fontSize: 15, fontWeight: FontWeight.w800),
                  ),
                  style: ElevatedButton.styleFrom(
                    backgroundColor: AppColors.primary,
                    foregroundColor: Colors.black,
                    padding: const EdgeInsets.symmetric(horizontal: 22, vertical: 14),
                    shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(40)),
                    elevation: 6,
                    shadowColor: AppColors.primary.withValues(alpha: 0.4),
                  ),
                ),
              ),
            ),
          ),

          // --- Details card (overlay when pin/list item tapped) ---
          if (_selected != null)
            Positioned(
              left: 0, right: 0, bottom: 0,
              child: SafeArea(
                top: false,
                child: PostDetailsCard(
                  post: _selected!,
                  onJoin: () => _join(_selected!),
                  onClose: () => setState(() => _selected = null),
                ),
              ),
            ),
        ],
      ),
    );
  }
```
